cmake_minimum_required(VERSION 3.16)
project(librw5100 VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
include(GNUInstallDirs)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(LIBUSB QUIET IMPORTED_TARGET libusb-1.0)
endif()
if(TARGET PkgConfig::LIBUSB)
  set(USB_TARGET PkgConfig::LIBUSB)
else()
  find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0 REQUIRED)
  find_library(LIBUSB_LIBRARY NAMES usb-1.0 libusb-1.0 REQUIRED)
  add_library(libusb_dep UNKNOWN IMPORTED)
  set_target_properties(libusb_dep PROPERTIES IMPORTED_LOCATION "${LIBUSB_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${LIBUSB_INCLUDE_DIR}")
  set(USB_TARGET libusb_dep)
endif()
add_library(rw5100 src/device.c src/transport.c src/frame.c src/card.c src/apdu.c)
add_library(rw5100::rw5100 ALIAS rw5100)
target_include_directories(rw5100 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
target_link_libraries(rw5100 PRIVATE ${USB_TARGET})
set_target_properties(rw5100 PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON
  VERSION ${PROJECT_VERSION} SOVERSION 0)
if(NOT WIN32)
  target_compile_definitions(rw5100 PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(rw5100 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(rw5100ctl tools/rw5100ctl.c)
target_link_libraries(rw5100ctl PRIVATE rw5100)
if(BUILD_TESTING)
  add_executable(test_frame tests/test_frame.c src/frame.c)
  target_include_directories(test_frame PRIVATE src include)
  add_test(NAME frame COMMAND test_frame)
  add_executable(test_transport tests/test_transport.c src/transport.c src/frame.c)
  target_include_directories(test_transport PRIVATE src include)
  target_link_libraries(test_transport PRIVATE ${USB_TARGET})
  if(NOT WIN32)
    target_compile_definitions(test_transport PRIVATE _POSIX_C_SOURCE=200809L)
  endif()
  add_test(NAME transport COMMAND test_transport)
  add_executable(test_card tests/test_card.c src/card.c src/frame.c)
  target_include_directories(test_card PRIVATE src include)
  add_test(NAME card COMMAND test_card)
  add_executable(test_apdu tests/test_apdu.c src/apdu.c)
  target_include_directories(test_apdu PRIVATE src include)
  add_test(NAME apdu COMMAND test_apdu)
  add_test(NAME cli_help COMMAND rw5100ctl --help)
endif()
install(TARGETS rw5100 rw5100ctl ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/rw5100.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

configure_file(librw5100.pc.in librw5100.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/librw5100.pc
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
